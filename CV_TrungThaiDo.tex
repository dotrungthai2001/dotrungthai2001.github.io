\cvsection{RESEARCH OVERVIEW}

\noindent{\bf Large-scale reconstruction methods for 3D photoacoustic tomography (PAT)} \\
\emph{PhD thesis --- IMT Toulouse \& A*STAR / IPAL Singapore, 2024--2028. Supervisors: C. Chaux, P. Escande, H. K. Lee.}
\begin{itemize}[leftmargin=*, noitemsep, topsep=2pt]
    \item Designing fast, accurate reconstruction methods --- combining inverse problems, optimization, and deep learning --- to deploy a novel 3D PAT scanner for biomedical imaging.
    \item Developed a combined Total Variation--Cauchy regularization for PAT image reconstruction (published at GRETSI 2025).
    \item Modeling the electrical impulse response of the 3D PAT system and implementing efficient forward/adjoint operators for large-scale data.
    \item In collaboration with the A*STAR team, exploring deep learning--based reconstruction for large-scale 3D PAT data.
\end{itemize}

\medskip
\noindent{\bf Inverse problems in photoacoustic tomography} \\
\emph{Research internship --- IMT Toulouse, France, 2023. Supervisor: P. Escande.}
\begin{itemize}[leftmargin=*, noitemsep, topsep=2pt]
    \item Implemented optimization algorithms to solve inverse problems in PAT.
    \item Built a blood-vessel image simulator for synthetic training data and trained deep neural networks for reconstruction (PyTorch).
\end{itemize}

\medskip
\noindent{\bf Computer vision for road-safety analysis} \\
